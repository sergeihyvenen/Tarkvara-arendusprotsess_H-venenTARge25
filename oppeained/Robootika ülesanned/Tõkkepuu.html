#include Servo.h 

const int trigPin = 9;
const int echoPin = 10;

const int servoPin = 6;

const int redPin = 3;
const int greenPin = 4;
const int bluePin = 5;

Servo barrier;

int openAngle = 90;
int closedAngle = 0;

int distanceLimit = 15; // cm

void setup()
{
  Serial.begin(9600);

  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);

  barrier.attach(servoPin);
  barrier.write(closedAngle);

  setColor(0, 0, 255); // sinine = ootab
}

void loop()
{
  int distance = getDistance();

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distance > 0 && distance < distanceLimit)
  {
    setColor(0, 255, 0); // roheline
    barrier.write(openAngle);
    delay(3000);
  }
  else
  {
    setColor(0, 0, 255); // sinine
    barrier.write(closedAngle);
  }

  delay(300);
}

int getDistance()
{
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH);
  int distance = duration * 0.034 / 2;

  return distance;
}

void setColor(int red, int green, int blue)
{
  analogWrite(redPin, red);
  analogWrite(greenPin, green);
  analogWrite(bluePin, blue);
}

Projekti nimi: Nutikas parkimistõkkepuu süsteem

Projekti eesmärk

Projekti eesmärk oli luua automaatne parkimistõkkepuu, mis tuvastab sõiduki ultrahelianduri abil ning avab tõkkepuu automaatselt servomootori abil. Süsteem kasutab RGB LED-i, et anda kasutajale visuaalset tagasisidet süsteemi oleku kohta.

Kasutatud komponendid

  • Arduino Uno R3
  • HC-SR04 ultraheliandur
  • Servomootor SG90
  • RGB LED
  • 220 Ω takistid
  • Breadboard
  • Ühendusjuhtmed
  • USB-kaabel

Süsteemi tööpõhimõte

  1. Ultraheliandur mõõdab pidevalt kaugust objektini.
  2. Kui sõiduk on lähemal kui 15 cm, tuvastatakse selle saabumine.
  3. Servomootor pöörab tõkkepuu 90° ja avab sissepääsu.
  4. RGB LED süttib roheliselt, mis näitab, et läbipääs on lubatud.
  5. Kui sõiduk eemaldub, sulgub tõkkepuu ning LED muutub siniseks.

Ühendused

  • HC-SR04: VCC → 5V, GND → GND, TRIG → D9, ECHO → D10
  • Servo: Signaal → D6, VCC → 5V, GND → GND
  • RGB LED: R → D3, G → D4, B → D5, ühine jalg → GND

Algoritm

  1. Süsteem käivitub.
  2. Arduino loeb ultrahelianduri mõõtmisi.
  3. Kui kaugus on väiksem kui 15 cm, avaneb tõkkepuu.
  4. Kui kaugus on suurem kui 15 cm, sulgub tõkkepuu.

Projekti eelised

  • Automaatne töö.
  • Lihtne ehitus ja programmeerimine.
  • Madal energiatarve.
  • Võimalik edasi arendada suuremaks süsteemiks.

Võimalikud edasiarendused

  • LCD-ekraan vabade parkimiskohtade kuvamiseks.
  • Teine ultraheliandur sõidukite loendamiseks.
  • Buzzer helisignaaliks.
  • RFID-kaardi või numbrituvastuse lisamine.
  • Wi-Fi moodul kaugjälgimiseks.

Kokkuvõte

Nutikas parkimistõkkepuu süsteem täidab edukalt oma eesmärgi. Süsteem tuvastab sõiduki, avab automaatselt tõkkepuu ning annab kasutajale visuaalset tagasisidet RGB LED-i abil. Projekti käigus õpiti kasutama Arduino mikrokontrollerit, ultraheliandurit, servomootorit ning looma automaatikasüsteemi.
