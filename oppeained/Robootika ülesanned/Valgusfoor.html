// VALGUSFOOR

// Autod
const int autoRoheline = 11;
const int autoKollane = 12;
const int autoPunane = 13;

// Jalakäija
const int jalaRoheline = 9;
const int jalaPunane = 10;

// Array for pins
const int pinsArray[] = {
  autoRoheline,
  autoKollane,
  autoPunane,
  jalaRoheline,
  jalaPunane
};

const int pinsArray_length =
sizeof(pinsArray) / sizeof(pinsArray[0]);

void setup()
{
  for (int i = 0; i < pinsArray_length; i++)
  {
    pinMode(pinsArray[i], OUTPUT);
  }
}

{
