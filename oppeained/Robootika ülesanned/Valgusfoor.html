// VALGUSFOOR

// Autod
const int autoRoheline = 11;
const int autoKollane = 12;
const int autoPunane = 13;

// Jalakäija
const int jalaRoheline = 9;
const int jalaPunane = 10;

// Array for pins
const int pinsArray[] = { autoRoheline, autoKollane, autoPunane, jalaRoheline, jalaPunane };
const int pinsArray_length = sizeof(pinsArray) / sizeof(pinsArray[0]);

// SETUP
void setup()
{
  for (int i = 0; i < pinsArray_length; i++) 
  {
    pinMode(pinsArray[i], OUTPUT);
  }
}

// LOOP
void loop() 
{
  paevReziim();
  delay(700);
  ooReziim();
}

// PÄEVAREŽIIM
void paevReziim()
{
  // Kõik OFF
  digitalWrite(autoRoheline, LOW);
  digitalWrite(autoKollane, LOW);
  digitalWrite(autoPunane, LOW);
  digitalWrite(jalaRoheline, LOW);
  digitalWrite(jalaPunane, LOW);
  
  // autoPunane ja jalaPunane -> ON
  digitalWrite(autoPunane, HIGH);
  digitalWrite(jalaPunane, HIGH);
  delay(2000);
  
  // jalaPunane -> jalaRoheline
  digitalWrite(jalaPunane, LOW);
  digitalWrite(jalaRoheline, HIGH);
  delay(2000);
  
  // blink + switch
  cycleBlink(jalaRoheline, jalaPunane);
  delay(2000);
  
  // autoPunane -> autoKollane
  digitalWrite(autoPunane, LOW);
  digitalWrite(autoKollane, HIGH);
  delay(2000);
  
  // autoKollane -> autoRoheline
  digitalWrite(autoKollane, LOW);
  digitalWrite(autoRoheline, HIGH);
  delay(2000);
  
  // blink + switch
  cycleBlink(autoRoheline, autoKollane);
  delay(2000);
  
  // autoKollane -> autoPunane
  digitalWrite(autoKollane, LOW);
  digitalWrite(autoPunane, HIGH);
  delay(2000);
}

// ÖÖREŽIIM
void ooReziim()
{
  // Kõik OFF
  digitalWrite(autoRoheline, LOW);
  digitalWrite(autoKollane, LOW);
  digitalWrite(autoPunane, LOW);
  digitalWrite(jalaRoheline, LOW);
  digitalWrite(jalaPunane, LOW);

  // autoKollane blink
  for (int i = 0; i < 10; i++)
  {
    blink(autoKollane);
  }
}

// Blink + switch
void cycleBlink(int colorToBlink, int colorToSwitchTo)
{
  blink(colorToBlink);
  digitalWrite(colorToBlink, LOW);
  digitalWrite(colorToSwitchTo, HIGH);
  delay(2000);
}

// Blink function
void blink(int color)
{
  for (int i = 0; i < 3; i++)
  {
    digitalWrite(color, LOW);
    delay(500);
    digitalWrite(color, HIGH);
    delay(500);
  }
}
