const int autopunane = 13;
const int autokollane = 12;
const int autoroheline = 11;

const int jalapunane = 6;
const int jalaroheline = 5;

const int sensorPin = A0;

int lightLevel;
const int threshold = 100;

void setup()
{
  pinMode(autopunane, OUTPUT);
  pinMode(autokollane, OUTPUT);
  pinMode(autoroheline, OUTPUT);

  pinMode(jalapunane, OUTPUT);
  pinMode(jalaroheline, OUTPUT);

  Serial.begin(9600);
}

void day()
{
  // Day mode

  digitalWrite(jalapunane, HIGH);
  digitalWrite(autokollane, LOW);
  digitalWrite(autopunane, HIGH);

  delay(1000);

  digitalWrite(jalapunane, LOW);
  digitalWrite(jalaroheline, HIGH);

  delay(3000);

  for (int i = 0; i <= 3; i++)
  {
    digitalWrite(jalaroheline, LOW);
    delay(500);

    digitalWrite(jalaroheline, HIGH);
    delay(500);
  }

  digitalWrite(jalaroheline, LOW);
  digitalWrite(jalapunane, HIGH);

  delay(1000);

  digitalWrite(autokollane, HIGH);

  delay(1000);

  digitalWrite(autopunane, LOW);
  digitalWrite(autokollane, LOW);
  digitalWrite(autoroheline, HIGH);

  delay(3000);

  for (int i = 0; i <= 3; i++)
  {
    digitalWrite(autoroheline, LOW);
    delay(500);

    digitalWrite(autoroheline, HIGH);
    delay(500);
  }

  digitalWrite(autoroheline, LOW);
  digitalWrite(autokollane, HIGH);

  delay(1000);
}

void night()
{
  // Night mode

  digitalWrite(autopunane, LOW);
  digitalWrite(autoroheline, LOW);

  digitalWrite(jalapunane, LOW);
  digitalWrite(jalaroheline, LOW);

  digitalWrite(autokollane, LOW);
  delay(500);

  digitalWrite(autokollane, HIGH);
  delay(500);
}

void loop()
{
  lightLevel = analogRead(sensorPin);

  Serial.print("Light level: ");
  Serial.println(lightLevel);

  if (lightLevel > threshold)
  {
    night();
  }
  else
  {
    day();
  }
}
