// DISTANCE VARIABLES
const int trigPin = 3;
const int echoPin = 2;

long duration;
int dist_result;

// MOTORS VARIABLES
const int mot1f = 6;
const int mot1b = 5;
const int mot2f = 11;
const int mot2b = 10;

int mot_speed = 225;
int k = 0;

// LOGIC VARIABLES
const int dist_stop = 25;
const int max_range = 800;
const int min_range = 0;

const int errorLED = 13;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(errorLED, OUTPUT);

  pinMode(mot1f, OUTPUT);
  pinMode(mot1b, OUTPUT);
  pinMode(mot2f, OUTPUT);
  pinMode(mot2b, OUTPUT);

  motors_stop();
}

void loop() {
  delay(100);

  int result = ping();

  if (result <= min_range || result >= max_range) {
    digitalWrite(errorLED, HIGH);
    motors_stop();
    delay(500);
    return;
  }

  if (result <= dist_stop) {
    digitalWrite(errorLED, LOW);

    motors_back();
    delay(1000);

    motors_stop();
    delay(200);

    motors_left();
    delay(300);

    motors_stop();
    delay(200);
  } else {
    motors_forward();
    delay(100);
  }
}

// CHECK DISTANCE FUNCTION
int ping() {
  int d1 = singlePing();
  delay(10);
  int d2 = singlePing();
  delay(10);
  int d3 = singlePing();

  dist_result = (d1 + d2 + d3) / 3;
  return dist_result;
}

int singlePing() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH, 30000);

  if (duration == 0) {
    return max_range;
  }

  return duration / 58;
}

void motors_forward() {
  analogWrite(mot1f, mot_speed);
  analogWrite(mot2f, mot_speed);
  digitalWrite(mot1b, LOW);
  digitalWrite(mot2b, LOW);
}

void motors_back() {
  digitalWrite(mot1f, LOW);
  digitalWrite(mot2f, LOW);
  analogWrite(mot1b, mot_speed);
  analogWrite(mot2b, mot_speed);
}

void motors_stop() {
  digitalWrite(mot1f, LOW);
  digitalWrite(mot2f, LOW);
  digitalWrite(mot1b, LOW);
  digitalWrite(mot2b, LOW);
}

void motors_left() {
  analogWrite(mot1f, mot_speed);
  digitalWrite(mot2f, LOW);
  digitalWrite(mot1b, LOW);
  analogWrite(mot2b, mot_speed);
}

void motors_right() {
  digitalWrite(mot1f, LOW);
  analogWrite(mot2f, mot_speed);
  analogWrite(mot1b, mot_speed);
  digitalWrite(mot2b, LOW);
}

void motors_forward_left() {
  k = mot_speed * 0.8;
  analogWrite(mot1f, mot_speed);
  analogWrite(mot2f, k);
  digitalWrite(mot1b, LOW);
  digitalWrite(mot2b, LOW);
}

void motors_forward_right() {
  k = mot_speed * 0.8;
  analogWrite(mot1f, k);
  analogWrite(mot2f, mot_speed);
  digitalWrite(mot1b, LOW);
  digitalWrite(mot2b, LOW);
}

void motors_back_left() {
  k = mot_speed * 0.8;
  digitalWrite(mot1f, LOW);
  digitalWrite(mot2f, LOW);
  analogWrite(mot1b, k);
  analogWrite(mot2b, mot_speed);
}

void motors_back_right() {
  k = mot_speed * 0.8;
  digitalWrite(mot1f, LOW);
  digitalWrite(mot2f, LOW);
  analogWrite(mot1b, mot_speed);
  analogWrite(mot2b, k);
}

Dokumentatsioon

Projekti nimi: Nutikas takistuste vältimise robot

Projekti eesmärk

Projekti eesmärk oli luua autonoomne robot, mis liigub iseseisvalt ning väldib takistusi ultrahelianduri abil. Robot kasutab Arduino Uno mikrokontrollerit, HC-SR04 ultraheliandurit ja L298N mootoridraiverit, et juhtida kahte alalisvoolumootorit. Süsteem võimaldab robotil automaatselt tuvastada takistusi ning muuta liikumissuunda ilma inimese sekkumiseta.

Kasutatud komponendid

  • Arduino Uno R3
  • HC-SR04 ultraheliandur
  • L298N mootoridraiver
  • 2 × DC mootorit
  • LED indikaator (D13)
  • 9V patareitoide
  • Ühendusjuhtmed
  • Roboti alusraam

Süsteemi tööpõhimõte

  1. Ultraheliandur mõõdab pidevalt kaugust roboti ees oleva objektini.
  2. Arduino teeb kolm järjestikust mõõtmist ning arvutab nende keskmise väärtuse.
  3. Kui kaugus on suurem kui 25 cm, liigub robot otse edasi.
  4. Kui kaugus on väiksem kui 25 cm, peatub robot, liigub ühe sekundi tagasi ning pöörab vasakule.
  5. Pärast pööramist jätkab robot automaatselt liikumist edasi.
  6. Kui andur ei saa korrektset mõõtmist, peatatakse mootorid ning süttib veatuli.

Ühendused

  • HC-SR04: TRIG → D3, ECHO → D2, VCC → 5V, GND → GND
  • L298N: IN1 → D6, IN2 → D5, IN3 → D11, IN4 → D10
  • LED: D13
  • DC mootorid: ühendatud L298N väljunditesse OUT1–OUT4
  • Toide: Arduino Uno ja 9V patarei

Programmi töö

  1. Robot käivitub ning kõik mootorid peatatakse.
  2. Ultraheliandur mõõdab pidevalt ees oleva objekti kaugust.
  3. Funktsioon ping() teeb kolm mõõtmist ja arvutab nende keskmise väärtuse.
  4. Kui kaugus on suurem kui 25 cm, liigub robot edasi.
  5. Kui kaugus on väiksem või võrdne 25 cm, liigub robot tagasi, peatub ning pöörab vasakule.
  6. Pärast pööramist jätkab robot uuesti liikumist edasi.
  7. Kui mõõtmine ebaõnnestub või kaugus on väljaspool lubatud vahemikku, peatatakse robot ning LED süttib.

Kasutatud funktsioonid

  • ping() – teeb kolm mõõtmist ja tagastab keskmise kauguse.
  • singlePing() – teeb ühe ultraheli mõõtmise.
  • motors_forward() – robot liigub edasi.
  • motors_back() – robot liigub tagasi.
  • motors_stop() – peatab mõlemad mootorid.
  • motors_left() – robot pöörab vasakule.
  • motors_right() – robot pöörab paremale.
  • motors_forward_left() ja motors_forward_right() – sujuvad pöörded edasi liikudes.
  • motors_back_left() ja motors_back_right() – pöörded tagurdamisel.

Projekti eelised

  • Autonoomne liikumine ilma kasutaja sekkumiseta.
  • Automaatne takistuste tuvastamine ultrahelianduriga.
  • Täpsem mõõtmine kolme järjestikuse mõõtmise keskmise abil.
  • Lihtne ehitus ja programmeerimine.
  • Sobib Arduino ja robootika õppimiseks.

Võimalikud edasiarendused

  • Servo lisamine ultraheliandurile 180° ümbruse skaneerimiseks.
  • Takistuste vältimine nii vasakul kui paremal pool.
  • Joonejärgimise andurite lisamine.
  • LCD-ekraan roboti oleku ja kauguse kuvamiseks.
  • Buzzer helisignaali jaoks.
  • Laetava aku kasutamine pikemaks tööajaks.

Kokkuvõte

Projekt "Nutikas takistuste vältimise robot" täidab edukalt oma eesmärgi. Robot liigub iseseisvalt, mõõdab ultrahelianduri abil kaugust ning väldib automaatselt takistusi. Takistuse avastamisel liigub robot tagasi, pöörab vasakule ja jätkab liikumist. Vigaste mõõtmiste korral peatatakse mootorid ning LED annab veast märku. Projekti käigus omandati praktilisi teadmisi Arduino programmeerimisest, ultrahelianduri kasutamisest ja mootorite juhtimisest L298N draiveri abil.
